79#if defined(_MSC_VER)
80//
81// Direct Show のビデオデバイスの一覧を作る
82//
83// https://docs.microsoft.com/en-us/windows/win32/directshow/selecting-a-capture-device
84// https://www.geekpage.jp/programming/directshow/list-capture-device.php
85// http://www.antillia.com/sol9.2.0/4.25.html
86//
87#include <dshow.h>
88#pragma comment(lib, "strmiids")
89
90void getDirectShowList(std::vector<std::string>& list)
91{
92 // COM を開く
93 HRESULT hr{ CoInitialize(nullptr) };
94 if (FAILED(hr)) return;
95
96 // デバイスの列挙子を作成する
97 ICreateDevEnum* pDevEnum{ nullptr };
98 hr = CoCreateInstance(CLSID_SystemDeviceEnum, NULL, CLSCTX_INPROC_SERVER, IID_ICreateDevEnum,
99 reinterpret_cast<PVOID*>(&pDevEnum));
100
101 // 列挙子が作れなかったら戻る
102 if (FAILED(hr)) return;
103
104 // デバイスの列挙子の異名を作成する
105 IEnumMoniker* pEnumMoniker{ nullptr };
106 hr = pDevEnum->CreateClassEnumerator(CLSID_VideoInputDeviceCategory, &pEnumMoniker, 0);
107
108 // デバイスの列挙子はもういらないので開放する
109 pDevEnum->Release();
110
111 // 列挙子の異名が作れなかったら戻る
112 if (FAILED(hr)) return;
113
114 // 列挙子の異名が一つもなければ戻る
115 if (!pEnumMoniker) return;
116
117 // 列挙子の異名の取り出し先
118 IMoniker* pMoniker{ nullptr };
119
120 // 列挙子の異名を一つずつ取り出す
121 for (int i = 0; pEnumMoniker->Next(1, &pMoniker, nullptr) == S_OK; ++i)
122 {
123 // プロパティバッグの場所を取り出す
124 IPropertyBag* pPropertyBag;
125 hr = pMoniker->BindToStorage(0, 0, IID_IPropertyBag, reinterpret_cast<void**>(&pPropertyBag));
126
127 // プロパティバッグの場所が取り出せなかった次に行く
128 if (FAILED(hr))
129 {
130 pMoniker->Release();
131 continue;
132 }
133
134 // FriendlyName の格納場所
135 VARIANT friendlyName;
136 VariantInit(&friendlyName);
137
138 // FriendlyName を取得する
139 hr = pPropertyBag->Read(L"FriendlyName", &friendlyName, 0);
140
141 // Friendly Name が取得できたら
142 if (SUCCEEDED(hr))
143 {
144 // 表示名に番号を追加してデバイスの一覧に追加する
145 list.emplace_back(FAILED(hr) ? "Unknown" : TCharToUtf8(friendlyName.bstrVal) + "##" + std::to_string(i));
146
147 // FriendlyName の格納場所を消去する
148 VariantClear(&friendlyName);
149 }
150
151 // プロパティバッグを解放する
152 pMoniker->Release();
153 pPropertyBag->Release();
154 }
155
156 // デバイスの列挙子の異名を開放する
157 pEnumMoniker->Release();
158
159 // COM を閉じる
160 CoUninitialize();
161}

164// Media Foundation のビデオデバイスの一覧を作る
165//
166// https://docs.microsoft.com/ja-jp/windows/win32/medfound/audio-video-capture-in-media-foundation
167//
168#include <Mfidl.h>
169#include <Mfapi.h>
170#include <Mferror.h>
171#pragma comment(lib, "mf.lib")
172#pragma comment(lib, "mfplat.lib")
173
174void getMediaFoundationList(std::vector<std::string>& list)
175{
176 // 検索条件を保持する属性ストアを作成する
177 IMFAttributes* pConfig{ NULL };
178 HRESULT hr{ MFCreateAttributes(&pConfig, 1) };
179
180 // 属性ストアが作成できなかったら戻る
181 if (FAILED(hr)) return;
182
183 // ビデオキャプチャデバイスを要求する
184 hr = pConfig->SetGUID(MF_DEVSOURCE_ATTRIBUTE_SOURCE_TYPE,
185 MF_DEVSOURCE_ATTRIBUTE_SOURCE_TYPE_VIDCAP_GUID);
186
187 // ビデオキャプチャデバイスが要求に失敗したら
188 if (FAILED(hr))
189 {
190 // 属性ストアを開放して戻る
191 pConfig->Release();
192 return;
193 }
194
195 // キャプチャデバイスを列挙する
196 IMFActivate** ppDevices{ NULL };
197 UINT32 count{ 0 };
198 hr = MFEnumDeviceSources(pConfig, &ppDevices, &count);
199
200 // ビデオキャプチャデバイスが列挙できなかったら
201 if (FAILED(hr))
202 {
203 // 属性ストアを開放して戻る
204 pConfig->Release();
205 return;
206 }
207
208 // それぞれのビデオキャプチャデバイスについて
209 for (DWORD i = 0; i < count; i++)
210 {
211 // ビデオキャプチャデバイスの表示名を取得する
212 WCHAR* szFriendlyName{ NULL };
213 UINT32 cchName{ 0 };
214 hr = ppDevices[i]->GetAllocatedString(MF_DEVSOURCE_ATTRIBUTE_FRIENDLY_NAME,
215 &szFriendlyName, &cchName);
216
217 // 表示名が取得できたら
218 if (SUCCEEDED(hr))
219 {
220 // 表示名に番号を追加してデバイスの一覧に追加する
221 list.emplace_back(TCharToUtf8(szFriendlyName) + "##" + std::to_string(i));
222 }
223
224 // 表示名の格納場所を解放する
225 CoTaskMemFree(szFriendlyName);
226 }
227
228 // それぞれのビデオキャプチャデバイスについて
229 for (DWORD i = 0; i < count; i++)
230 {
231 // ビデオキャプチャデバイスを開放する
232 ppDevices[i]->Release();
233 }
234
235 // ビデオキャプチャデバイスのインターフェースに使ったメモリを解放する
236 CoTaskMemFree(ppDevices);
237}

262bool Menu::openDevice()
263{
264 // バックエンドが GStreamer なら
265 if (backend == cv::CAP_GSTREAMER)
266 {
267 // パイプライン設定の取り出し
268 const auto& pipeline{ config.gstreamerPipelines[deviceNumber] };
269
270 // ダイアログで指定したパイプラインが開けなかったら
271 if (!capture.openMovie(pipeline, backend))
272 {
273 // 開けなかった
274 errorMessage = u8"パイプラインが開けません";
275 return false;
276 }
277
278 // GStreamer が使える
279 return true;
280 }
281
282 // コーデック
283 char codec[5]{};
284 if (codecNumber > 0) strncpy(codec, codecList[codecNumber], 5);
285
286 // ダイアログで指定したキャプチャデバイスが開けなかったら
287 if (!capture.openDevice(deviceNumber,
288 intrinsics.size, intrinsics.fps, backend, codec))
289 {
290 // 開けなかった
291 errorMessage = u8"デバイスが開けません";
292 return false;
293 }
294
295 // 使うことになったコーデックの番号を調べる
296 for (size_t i = 0; i < codecList.size(); ++i)
297 {
298 if (strncmp(codec, codecList[i], 4) == 0)
299 {
300 // コーデックが分かった
301 codecNumber = static_cast<int>(i);
302 return true;
303 }
304 }
305
306 // コーデックが分からない
307 codecNumber = 0;
308 return true;
309}

354 const auto fileListLength{ static_cast<int>(fileHistory.size()) };
355
356 // ファイルのリストの各ファイルについて
357 for (deviceNumber = 0; deviceNumber < fileListLength; ++deviceNumber)
358 {
359 // 選択したファイルと同じものがあればそれを選択する
360 if (fileHistory[deviceNumber] == filepath) break;
361 }
362
363 // 選択したファイルがファイルのリストの中になければ
364 if (deviceNumber == fileListLength)
365 {
366 // その先頭にファイルパスを挿入して
367 fileHistory.insert(fileHistory.begin(), filepath);
368
369 // そのエントリを選択する
370 deviceNumber = 0;
371 }
372
373 // ダイアログで指定した動画ファイルが開けたら
374 if (capture.openMovie(filepath, backend))
375 {
376 // 構成データの解像度と画角を開いた画像に合わせる

505//
506// 較正用の画像ファイルを取得する (複数選択)
507//
508void Menu::recordFileCorners() const
509{
510 // ファイルダイアログから得るパス
511 const nfdpathset_t* outPaths;
512
513 // ファイルダイアログを開く
514 if (NFD_OpenDialogMultiple(&outPaths, imageFilter, 1, NULL) == NFD_OKAY)
515 {
516 // ファイルパスの一覧を得る
517 nfdpathsetenum_t enumerator;
518 NFD_PathSet_GetEnum(outPaths, &enumerator);
519
520 // ファイルパスの一覧からファイルパスを一つずつ取り出して
521 for (nfdchar_t* path = NULL; NFD_PathSet_EnumNext(&enumerator, &path) && path;)
522 {
523 // 画像の読み出し
524 CamImage image;
525
526 // 画像ファイルが読み出せたら
527 if (image.open(path))
528 {
529 // データのコピー先
530 cv::Mat frame;
531
532 // データをコピーして
533 image.transmit(frame);
534
535 // ボードを検出して
536 calibration.detectBoard(frame);
537
538 // コーナーを記録する
539 calibration.recordCorners();
540
541 // 画像の読み出しを終わる
542 image.close();
543 }
544
545 // ファイルパスの取り出しに使ったメモリを開放する
546 NFD_PathSet_FreePath(path);
547 }
548
549 // ファイルパスの一覧に使ったメモリを開放する
550 NFD_PathSet_FreeEnum(&enumerator);
551
552 // フォルダのパスに使ったメモリを開放する
553 NFD_PathSet_Free(outPaths);
554 }
555}
