83 createBoard(checkerSize, checkerLength);

97void Calibration::detectBoard(cv::Mat& image)
98{
99 // 画像のサイズを保存しておく
100 size = image.size();
101
102 // 4 チャンネル画像の場合は一時的に3チャンネル画像を作成する
103 cv::Mat tempImage;
104 const auto isFourChannels{ image.channels() == 4 };

122
123 // 4 チャンネル画像の場合は結果を書き戻す

130//
131// ArUco Marker を検出する
132//
133void Calibration::detectMarkers(cv::Mat& image, float markerLength)
134{
135 // 4 チャンネル画像の場合は一時的に3チャンネル画像を作成する
136 cv::Mat tempImage;
137 const auto isFourChannels{ image.channels() == 4 };
138 if (isFourChannels)
139 {
140 cv::cvtColor(image, tempImage, cv::COLOR_BGRA2BGR);
141 }
142 else
143 {
144 tempImage = image;
145 }
146
147 // ArUco Marker のコーナーを検出する
148 detector->detectMarkers(tempImage, corners, ids, rejected);
149
150 // コーナーが見つからなければ戻る
151 if (corners.empty()) return;
152
153 // キャリブレーションが完了していれば
154 if (finished())
155 {
156 // 各マーカに対応する３次元空間の点
157 const float markerCenter{ markerLength * 0.5f };
158 std::vector<cv::Point3f> markerObjPoints;
159 markerObjPoints.push_back(cv::Point3f(-markerCenter, markerCenter, 0.0f));
160 markerObjPoints.push_back(cv::Point3f(markerCenter, markerCenter, 0.0f));
161 markerObjPoints.push_back(cv::Point3f(markerCenter, -markerCenter, 0.0f));
162 markerObjPoints.push_back(cv::Point3f(-markerCenter, -markerCenter, 0.0f));
163
164 // 個々のマーカーについて
165 for (size_t i = 0; i < corners.size(); ++i)
166 {
167 // マーカーのコーナー検出位置から3次元姿勢（回転・平行移動）を推定する
168 cv::Vec3d rvec, tvec;
169 cv::solvePnP(markerObjPoints, corners[i], cameraMatrix, distCoeffs, rvec, tvec, false, cv::SOLVEPNP_IPPE_SQUARE);
170
171 // 座標軸を描く
172 cv::drawFrameAxes(tempImage, cameraMatrix, distCoeffs, rvec, tvec, markerLength);
173 }
174 }
175 else
176 {
177 // ArUco Marker の場所に矩形と番号を描き込む
178 cv::aruco::drawDetectedMarkers(tempImage, corners, ids);
179 }
180
181 // 4 チャンネル画像の場合は結果を書き戻す
182 if (isFourChannels)
183 {
184 cv::cvtColor(tempImage, image, cv::COLOR_BGR2BGRA);
185 }
186}

189// 標本を取得する
190//
191void Calibration::recordCorners()
192{
193 // ChArUco Board のコーナーが４つ以上見つかれば
194 if (charucoCorners.size() >= 4)
195 {
196 // ChArUco Board のレイアウトと検出されたコーナーから
197 // ChArUco Board 上の点と対応する画像上の点を求める
198 board->matchImagePoints(charucoCorners, charucoIds, objectPoints, imagePoints);
199
200 // ChArUco Board 上の点と対応する画像上の点が見つかれば
201 if (!imagePoints.empty() && !objectPoints.empty())
202 {
203 // ChArUco Board のコーナーを記録する
204 allCorners.push_back(charucoCorners);
205 allIds.push_back(charucoIds);
206 allImagePoints.push_back(imagePoints);
207 allObjectPoints.push_back(objectPoints);
208
209 // 記録したコーナーの数の合計を求める
210 totalCorners += static_cast<int>(charucoCorners.size());
211
212 // 記録したショットの幾何特徴を保存する (多様性チェック用)
213 lastRecordedFeatures = computeFeatures(charucoCorners);
214 hasRecordedShot = true;
215 stableDuration = 0.0f;
216 isCurrentlyStable = false;
217 }
218 }
219#if defined(_DEBUG)
220 std::cerr << "charucoCorners = " << charucoCorners.size()
221 << ", allCorners = " << allCorners.size() << "\n";
222#endif
223}

253//
254// 較正する
255//
256bool Calibration::calibrate()
257{
258 // 再投影誤差はとりあえず 0 にしておく
259 repError = 0.0f;
260
261 // コーナーを合計６つ以上検出できていれば
262 if (allCorners.size() >= 6) try
263 {
264 // CALIB_USE_INTRINSIC_GUESS が設定されていない場合に、
265 // fx と fy をしてしたアスペクト比に強制する
266 if (calibrationFlags & cv::CALIB_FIX_ASPECT_RATIO)
267 {
268 const auto aspect{ static_cast<double>(size.width) / size.height };
269 cameraMatrix = cv::Mat::eye(3, 3, CV_64F);
270 cameraMatrix.at<double>(0, 0) = aspect;
271 }
272
273 // ChArUco Board の姿勢 (使わないので捨ててしまう)
274 std::vector<cv::Mat> boardRvecs, boardTvecs;
275
276 // 取得した全てのコーナーからカメラパラメータを推定する
277 repError = cv::calibrateCamera(allObjectPoints, allImagePoints, size,
278 cameraMatrix, distCoeffs, boardRvecs, boardTvecs, cv::noArray(),
279 cv::noArray(), cv::noArray(), calibrationFlags);
280 }
281 catch (const cv::Exception&)
282 {
283 // 較正に失敗した場合は計算結果を捨てる
284 discardCorners();
285
286 // 較正に失敗したことを報告する
287 return false;
288 }
289
290 // 較正の計算結果を再利用する
291 calibrationFlags |= cv::CALIB_USE_INTRINSIC_GUESS;
292
293 // 較正に成功したことを報告する
294 return true;
295}

297//
298// 回転ベクトルから姿勢の変換行列を求める
299//
300GgMatrix Calibration::RvecTvecToPose(const cv::Vec3d& rvec, const cv::Vec3d& tvec)
301{
302#if defined(USE_RODRIGUES)
303 // 回転軸と回転角から回転の変換行列を求める
304 cv::Mat_<double> r(3, 3);
305 cv::Rodrigues(rvec, r);
306
307 // 姿勢の変換行列
308 return GgMatrix
309 {
310 static_cast<GLfloat>(r[0][0]),
311 static_cast<GLfloat>(r[1][0]),
312 static_cast<GLfloat>(r[2][0]),
313 0.0f,
314 static_cast<GLfloat>(r[0][1]),
315 static_cast<GLfloat>(r[1][1]),
316 static_cast<GLfloat>(r[2][1]),
317 0.0f,
318 static_cast<GLfloat>(r[0][2]),
319 static_cast<GLfloat>(r[1][2]),
320 static_cast<GLfloat>(r[2][2]),
321 0.0f,
322 static_cast<GLfloat>(tvec[0]),
323 static_cast<GLfloat>(tvec[1]),
324 static_cast<GLfloat>(tvec[2]),
325 1.0f
326 };
327#else
328 // 回転角
329 const auto d{ cv::norm(rvec) };
330
331 // 回転軸ベクトル
332 const auto rx{ static_cast<GLfloat>(rvec[0] / d) };
333 const auto ry{ static_cast<GLfloat>(rvec[1] / d) };
334 const auto rz{ static_cast<GLfloat>(rvec[2] / d) };
335
336 // 平行移動量
337 const auto tx{ static_cast<GLfloat>(tvec[0]) };
338 const auto ty{ static_cast<GLfloat>(tvec[1]) };
339 const auto tz{ static_cast<GLfloat>(tvec[2]) };
340
341 // 姿勢の変換行列
342 return ggTranslate(tx, ty, tz).rotate(rx, ry, rz, static_cast<GLfloat>(d));
343#endif
344}

371//
372// カメラパラメータの JSON オブジェクトから数値の配列を取得する
373//
374static bool getMatrix(const picojson::object& object,
375 const std::string& key, cv::Mat& mat, int cols, int rows)
376{
377 // key に一致するオブジェクトを探す
378 const auto&& value{ object.find(key) };
379
380 // オブジェクトが無いか配列でなかったら戻る
381 if (value == object.end() || !value->second.is<picojson::array>()) return false;
382
383 // 配列を取り出す
384 const auto& array{ value->second.get<picojson::array>() };
385
386 // 配列の要素数とデータの格納先の行列の要素数が一致していなければ戻る
387 if (static_cast<size_t>(cols) * rows != array.size()) return false;
388
389 // カメラ行列の要素を確保する
390 mat = cv::Mat::zeros(rows, cols, CV_64F);
391
392 // 配列の要素について
393 for (size_t i = 0; i < array.size(); ++i)
394 {
395 // 行列の要素の位置
396 const auto x{ static_cast<int>(i % cols) };
397 const auto y{ static_cast<int>(i / cols) };
398
399 // 要素が数値なら格納する
400 if (array[i].is<double>()) mat.at<double>(y, x) = array[i].get<double>();
401 }
402
403 return true;
404}

406//
407// カメラパラメータの JSON オブジェクトから数値の配列を取得する
408//
409static void setMatrix(picojson::object& object,
410 const std::string& key, const cv::Mat& mat)
411{
412 // picojson の配列
413 picojson::array array;
414
415 // 配列の要素について
416 for (size_t i = 0; i < mat.total(); ++i)
417 {
418 // 行列の要素の位置
419 const auto x{ static_cast<int>(i % mat.cols) };
420 const auto y{ static_cast<int>(i / mat.cols) };
421
422 // 要素を picojson::array に追加する
423 array.emplace_back(picojson::value(mat.at<double>(y, x)));
424 }
425
426 // オブジェクトに追加する
427 object.emplace(key, array);
428}

430//
431// ファイルからキャリブレーションパラメータを読み込む
432//
433bool Calibration::loadParameters(const std::string& filename)
434{
435 // パラメータファイルの読み込み
436 std::ifstream json{ Utf8ToTChar(filename) };
437 if (!json) return false;
438
439 // JSON の読み込み
440 picojson::value value;
441 json >> value;
442 json.close();
443
444 // 構成内容の取り出し
445 const auto& object{ value.get<picojson::object>() };
446
447 // オブジェクトが空だったらエラー
448 if (object.empty()) return false;
449
450 // カメラ行列
451 if (!getMatrix(object, "camera matrix", cameraMatrix, 3, 3)) return false;
452
453 // 歪み定数
454 if (!getMatrix(object, "distortion", distCoeffs, 5, 1)) return false;
455
456 // 再投影誤差
457 getValue(object, "error", repError);
458
459 // 較正の計算結果を再利用しない
460 calibrationFlags &= ~cv::CALIB_USE_INTRINSIC_GUESS;
461
462 // キャリブレーションパラメータの読み込み
463 return true;
464}

522//
523// コーナー群から幾何特徴を計算する
524//
525// 【目的】
526// ChArUco ボードの画像内での配置状態（位置、見かけの大きさ、傾き）を
527// 統計的モーメントを用いてロバストに要約し、特徴量として抽出する。
528// 直前に記録した標本の特徴量と比較することで、サンプルの多様性（位置移動、
529// カメラとの距離変化、回転・傾き）を判定するために利用する。
530//
531Calibration::BoardPoseFeatures Calibration::computeFeatures(const std::vector<cv::Point2f>& corners)
532{
533 BoardPoseFeatures features;
534
535 // 4点未満では面としての幾何特徴（慣性楕円など）を安定して算出できないため除外
536 if (corners.size() >= 4)
537 {
538 // コーナー点群の統計モーメントを算出
539 // m00: 点の総数 (質量)
540 // m10, m01: 1次モーメント (座標総和)
541 // mu20, mu02, mu11: 重心まわりの2次中心モーメント (分散・共分散)
542 cv::Moments m = cv::moments(corners);
543 if (m.m00 > 0.0)
544 {
545 // 1. 重心位置 (Centroid): 画像内でのボードの中心座標
546 features.centroid = cv::Point2f(static_cast<float>(m.m10 / m.m00), static_cast<float>(m.m01 / m.m00));
547
548 const double mu20 = m.mu20 / m.m00;
549 const double mu02 = m.mu02 / m.m00;
550 const double mu11 = m.mu11 / m.m00;
551
552 // 2. 慣性半径 (Spread / Scale): コーナー点群の重心からの平均的な広がり
553 // カメラとボード間の距離（見かけの大きさ）を表す指標となる。
554 // 単純な外接矩形や一部コーナーのオクルージョンに比べ外れ値に強い。
555 features.spread = static_cast<float>(std::sqrt(std::max(0.0, mu20 + mu02)));
556
557 // 3. 主軸角度 (Orientation Angle): 慣性主軸の傾き角度 (度数法, -90° ～ +90°)
558 // 画像平面内におけるボードの回転・傾き状態を表す指標となる。
559 features.angleDeg = static_cast<float>(0.5 * std::atan2(2.0 * mu11, mu20 - mu02) * 180.0 / CV_PI);
560 }
561 }
562 return features;
563}

565//
566// モーション状態（静止判定）を更新する
567//
568// 【目的】
569// 直前フレームと同一の ChArUco コーナー ID を照合してフレーム間の変位量を追跡し、
570// 作業者の手振れやボード移動に伴う「モーションブラー（ブレ）」のない
571// 安定した静止状態が一定時間継続しているかを判定する。
572//
573void Calibration::updateMotion(float deltaTime, float motionThresholdPx, float minStableTime, int minCorners)
574{
575 // 検出コーナー数が最低必要数未満、または直前フレームのデータが存在しない場合
576 if (charucoCorners.size() < static_cast<size_t>(minCorners) || prevCharucoCorners.empty())
577 {
578 // 移動中または未検出とみなし、変位量を大きく設定して静止時間をリセット
579 currentMotion = 999.0f;
580 stableDuration = 0.0f;
581 isCurrentlyStable = false;
582 prevCharucoCorners = charucoCorners;
583 prevCharucoIds = charucoIds;
584 return;
585 }
586
587 // 直前フレームと現在フレームで共通するコーナー ID を探索し、
588 // ピクセル座標の移動量（ユークリッド距離）を合算する
589 float totalDist = 0.0f;
590 int commonCount = 0;
591
592 for (size_t i = 0; i < charucoIds.size(); ++i)
593 {
594 const int id = charucoIds[i];
595 for (size_t j = 0; j < prevCharucoIds.size(); ++j)
596 {
597 if (prevCharucoIds[j] == id)
598 {
599 totalDist += static_cast<float>(cv::norm(charucoCorners[i] - prevCharucoCorners[j]));
600 ++commonCount;
601 break;
602 }
603 }
604 }
605
606 // 共通コーナーが一定数以上見つかった場合は平均変位量 (px) を算出
607 if (commonCount >= std::min(4, minCorners))
608 {
609 currentMotion = totalDist / commonCount;
610 }
611 else
612 {
613 // 共通点が少なすぎる場合はトラッキング不可（大きな移動があった）とみなす
614 currentMotion = 999.0f;
615 }
616
617 // 平均変位量が許容閾値（motionThresholdPx、例: 2.0px）以下なら静止中と判定
618 if (currentMotion <= motionThresholdPx)
619 {
620 // 静止継続時間 (秒) を積算
621 stableDuration += deltaTime;
622 }
623 else
624 {
625 // 動いている場合は静止タイマーを即座にリセット
626 stableDuration = 0.0f;
627 }
628
629 // 静止継続時間が必要時間（minStableTime、例: 0.6秒）に達していれば安定と判定
630 isCurrentlyStable = (stableDuration >= minStableTime);
631
632 // 次フレームの変位計算用に現在のコーナー情報を保持
633 prevCharucoCorners = charucoCorners;
634 prevCharucoIds = charucoIds;
635}
636
637//
638// 直前に記録された標本に対して十分な姿勢・位置の多様性があるかを調べる
639//
640// 【目的】
641// 静止が検知された場合でも、直前に記録したサンプルとほぼ同じ場所・距離・角度の
642// 重複標本が連続して保存されるのを防止する。
643// 画面内の平行移動、カメラとの距離変化、ボードの傾き変化のいずれかが
644// 閾値以上ある場合のみ「多様性あり」と判定して記録を許可する。
645//
646bool Calibration::isDiverseEnough(float minDistanceRatio, float minScaleRatio, float minAngleDeg) const
647{
648 // 初回記録時、または過去の標本がない場合は比較対象がないため常に合格
649 if (!hasRecordedShot || allCorners.empty()) return true;
650
651 // コーナーが少なすぎる場合は判定不可
652 if (charucoCorners.size() < 4) return false;
653
654 // 現在フレームの幾何特徴を算出
655 const auto cur = computeFeatures(charucoCorners);
656
657 // 1. 重心移動量の比率 (画像対角線長に対する割合)
658 // ボードが画面の中央、四隅、端など異なる位置に移動したかを判定
659 const float diag = std::hypot(static_cast<float>(size.width), static_cast<float>(size.height));
660 const float dist = static_cast<float>(cv::norm(cur.centroid - lastRecordedFeatures.centroid));
661 if (diag > 0.0f && (dist / diag) >= minDistanceRatio) return true;
662
663 // 2. サイズ（慣性半径・スケール）の変化率
664 // カメラに近づいたか、遠ざかったか（奥行き方向のバリエーション）を判定
665 if (lastRecordedFeatures.spread > 0.0f)
666 {
667 const float scaleDiff = std::abs(cur.spread - lastRecordedFeatures.spread) / lastRecordedFeatures.spread;
668 if (scaleDiff >= minScaleRatio) return true;
669 }
670
671 // 3. 主軸角度の変化 (度)
672 // ボードが回転または斜めに傾けられたかを判定 (周期性を考慮して 0°～90° の最小差を算出)
673 float angleDiff = std::abs(cur.angleDeg - lastRecordedFeatures.angleDeg);
674 while (angleDiff > 90.0f) angleDiff = std::abs(180.0f - angleDiff);
675 if (angleDiff >= minAngleDeg) return true;
676
677 // いずれの幾何変化も閾値に満たない場合は「同一姿勢の重複」とみなして不合格
678 return false;
679}
