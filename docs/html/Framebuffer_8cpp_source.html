110void Framebuffer::create(GLsizei width, GLsizei height, int channels)
111{
112 // 既存のテクスチャを破棄して新しいテクスチャを作成する
113 Texture::create(width, height, channels);
114
115 // まだメッシュの頂点配列オブジェクトが作られていなければ作る
116 if (!mesh) mesh = std::make_shared<Mesh>();
117
118 // 指定したサイズがフレームバッファオブジェクトのサイズと同じなら何もしない
119 if (width == framebufferSize[0] && height == framebufferSize[1]
120 && channels == framebufferChannels) return;
121
122 // フレームバッファオブジェクトのサイズとチャンネル数を記録する
123 framebufferSize = std::array<int, 2>{ width, height };
124 framebufferChannels = channels;
125
126 // 以前のフレームバッファオブジェクトを削除する
127 glDeleteFramebuffers(1, &framebufferName);
128
129 // フレームバッファオブジェクトを作成する
130 glGenFramebuffers(1, &framebufferName);
131 glBindFramebuffer(GL_FRAMEBUFFER, framebufferName);
132 glFramebufferTexture2D(GL_FRAMEBUFFER, attachment, GL_TEXTURE_2D, getTextureName(), 0);
133 glDrawBuffers(1, &attachment);
134 glReadBuffer(attachment);
135 glBindFramebuffer(GL_FRAMEBUFFER, 0);
136}
