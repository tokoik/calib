125 void capture()
126 {
127 // 再生開始時刻
128 auto startTime{ getTime() };
129
130 // スレッドが実行可の間
131 while (running)
132 {
133 // フレームを取り出せたら true
134 auto status{ (total <= 0.0 || camera.get(cv::CAP_PROP_POS_FRAMES) < out) && camera.grab() };
135
136 // ムービーファイルでないかムービーファイルの終端でなければ次のフレームを取り出して
137 if (status && camera.retrieve(cvFrame))
138 {
139 // 単一バッファをロックしてから
140 std::lock_guard<std::mutex> lock{ mtx };
141
142 // 基底クラスのバッファとメンバを直接更新
143 width = cvFrame.cols;
144 height = cvFrame.rows;
145 channels = cvFrame.channels();
146 const auto size{ static_cast<size_t>(cvFrame.total() * cvFrame.elemSize()) };
147 image.resize(size);
148 std::memcpy(image.data(), cvFrame.data, size);
149
150 // 新しいフレームがキャプチャされたことを通知する
151 captured = true;
152 }
153
154 // 遅延時間
155 auto deferred{ 0.0 };
156
157 // ムービーファイルから入力しているとき
158 if (total > 0.0)
159 {
160 // ムービーファイルの終端に到達していたら
161 if (!status)
162 {
163 // インポイントまで巻き戻す
164 camera.set(cv::CAP_PROP_POS_FRAMES, in);
165
166 // 経過時間を戻す
167 elapsedTime = 0.0;
168
169 // 開始時間を更新する
170 startTime = getTime();
171 }
172 else
173 {
174 // 現在のフレームのインポイントからの経過時間
175 const auto pos{ camera.get(cv::CAP_PROP_POS_MSEC) - in };
176
177 // 再生位置の次のフレームの時刻に対する経過時間
178 const auto now{ (elapsedTime + getTime() - startTime) * 1000.0 + interval };
179
180 // 遅延時間はフレームの経過時間と現在の経過時間の差
181 deferred = pos - now;
182 }
183 }
184
185 // 遅延時間あれば待つ
186 if (deferred > 0.0)
187 {
188 std::this_thread::sleep_for(std::chrono::milliseconds(static_cast<int>(deferred)));
189 }
190 }
191
192 // 再生停止までの経過時間を積算する
193 elapsedTime += getTime() - startTime;
194 }
