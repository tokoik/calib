89bool Config::load(const pathString& filename)
90{
91 // 構成ファイルの読み込み
92 std::ifstream json{ filename };
93 if (!json) return false;
94
95 // JSON の読み込み
96 picojson::value value;
97 json >> value;
98 json.close();
99
100 // ルートが JSON オブジェクトでなければ、既存構成を変更せず失敗とする
101 if (!value.is<picojson::object>()) return false;
102
103 // 構成内容の取り出し
104 const auto& object{ value.get<picojson::object>() };
105
106 // オブジェクトが空だったらエラー
107 if (object.empty()) return false;
108
109 // 読み込みに失敗しても現在の構成を壊さないよう、一時領域へ読み込む
110 auto loadedWindowSize{ windowSize };
111 auto loadedBackground{ background };
112 auto loadedSettings{ settings };
113 auto loadedInitialImage{ initialImage };
114 std::vector<Preference> loadedPreferences;
115
116 // ウィンドウのサイズ
117 getValue(object, "size", loadedWindowSize);
118
119 // ウィンドウの背景色
120 getValue(object, "background", loadedBackground);
121
122 // メッシュのサンプル数
123 getValue(object, "samples", loadedSettings.samples);
124
125 // キャプチャデバイスの姿勢
126 getValue(object, "pose", loadedSettings.euler);
127
128 // 描画時の焦点距離
129 getValue(object, "focal", loadedSettings.focal);
130
131 // 描画時の焦点距離の範囲
132 getValue(object, "range", loadedSettings.focalRange);
133
134 // ArUco Marker の辞書名
135 getString(object, "dictionary", loadedSettings.dictionaryName);
136
137 // 初期表示画像
138 getString(object, "initial", loadedInitialImage);
139
140 // キャプチャデバイスの構成を探す
141 const auto& camera{ object.find("camera") };
142
143 // キャプチャデバイスの構成の配列が見つからなければエラー
144 if (camera == object.end() || !camera->second.is<picojson::array>()) return false;
145
146 // 配列の個々の要素について
147 for (const auto& value : camera->second.get<picojson::array>())
148 {
149 // 配列中に投影方式以外の値があれば、不完全な構成を適用せず失敗とする
150 if (!value.is<picojson::object>()) return false;
151
152 // キャプチャデバイスの構成のオブジェクトを取り出す
153 const auto& preference{ value.get<picojson::object>() };
154
155 // キャプチャデバイスの構成をリストに追加
156 loadedPreferences.emplace_back(preference);
157 }
158
159 // キャプチャデバイスの構成が一つも読み取れなければエラー
160 if (loadedPreferences.empty()) return false;
161
162 // 実行時の再読み込みでは、新しい構成のシェーダも準備する
163 if (initialized)
164 {
165 // OpenGL 初期化後の再読み込みなので、置換前に全投影方式を描画可能にする
166 for (auto& preference : loadedPreferences) preference.buildShader();
167 }
168
169 // 読み込みと検証がすべて成功してから現在の構成を置き換える
170 windowSize = loadedWindowSize;
171 background = loadedBackground;
172 settings = loadedSettings;
173 initialImage = std::move(loadedInitialImage);
174 preferenceList.swap(loadedPreferences);
175
176 if (initialized)
177 {
178 // 実行中の再読み込みでは、新しい背景色を現在の OpenGL 状態にも反映する
179 glClearColor(background[0], background[1], background[2], background[3]);
180 }
181
182 return true;
183}
