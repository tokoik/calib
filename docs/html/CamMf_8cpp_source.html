84//
85// 初期化
86//
87const char* CamMf::ComInitializer::initialize()
88{
89 // エラーメッセージ
90 const char* message{ nullptr };
91
92 // 検索条件を保持する属性ストア
93 IMFAttributes* pAttributes{ nullptr };
94
95 // COM ライブラリを初期化する
96 if (FAILED(CoInitializeEx(nullptr, COINIT_MULTITHREADED)))
97 {
98 // COM ライブラリの初期化に失敗した
99 message = "Failed to initialize COM library.";
100 goto done;
101 }
102
103 // COM ライブラリの初期化に成功した
104 coInitialized = true;
105
106 // Media Foundation を起動する
107 if (FAILED(MFStartup(MF_VERSION, MFSTARTUP_FULL)))
108 {
109 // Media Foundation の起動に失敗した
110 message = "Failed to start Media Foundation.";
111 goto done;
112 }
113
114 // Media Foundation の起動に成功した
115 mfStarted = true;
116
117 // 検索条件を保持する属性ストアを作成する
118 if (FAILED(MFCreateAttributes(&pAttributes, 1)))
119 {
120 // 属性ストアの作成失敗した
121 message = "Failed to create attribute store.";
122 goto done;
123 }
124
125 // 属性ストアにビデオキャプチャデバイスの属性を設定する
126 if (FAILED(pAttributes->SetGUID(
127 MF_DEVSOURCE_ATTRIBUTE_SOURCE_TYPE,
128 MF_DEVSOURCE_ATTRIBUTE_SOURCE_TYPE_VIDCAP_GUID)))
129 {
130 // ビデオキャプチャデバイスの属性の設定に失敗した
131 message = "Failed to set attribute for video capture device.";
132 goto done;
133 }
134
135 // メディアソースを列挙する
136 if (FAILED(MFEnumDeviceSources(pAttributes,
137 &ppSourceActivate, &cSourceActivate)))
138 {
139 // メディアソースの列挙に失敗した
140 message = "Failed to enumerate media sources.";
141 goto done;
142 }
143
144 // すべてのメディアソースについて
145 for (DWORD i = 0; i < cSourceActivate; ++i)
146 {
147 // メディアソースの表示名の文字列ポインタ
148 WCHAR* szFriendlyName{ nullptr };
149
150 // メディアソースの表示名の文字数
151 UINT32 cFriendlyName{ 0 };
152
153 // メディアソースの表示名を取得する
154 if (SUCCEEDED(ppSourceActivate[i]->GetAllocatedString(
155 MF_DEVSOURCE_ATTRIBUTE_FRIENDLY_NAME, &szFriendlyName, &cFriendlyName)))
156 {
157 // ビデオキャプチャデバイス名を作る
158 std::stringstream ss;
159 ss << TCharToUtf8(szFriendlyName) << "##" << i;
160
161 // ビデオキャプチャデバイス名をリストに追加する
162 deviceList.emplace_back(ss.str());
163 }
164
165 // 表示名の文字列に使ったメモリを解放する
166 CoTaskMemFree(szFriendlyName);
167 }
168
169done:
170
171 // 属性ストアはもう使わないので解放する
172 SafeRelease(&pAttributes);
173
174 // エラーメッセージを返す
175 return message;
176}

217//
218// 使用可能な解像度、フレームレート、コーデックのリストを作成する
219//
220bool CamMf::enumerateFormats()
221{
222 // Source Reader が作成されていなければ戻る
223 if (!pSourceReader) return false;
224
225 // 以前に作成したリストをクリアする
226 availableFormats.clear();
227 formatList.clear();
228
229 // ネイティブメディアタイプを一つずつ列挙する
230 IMFMediaType* pMediaType{ nullptr };
231 for (DWORD dwMediaTypeIndex = 0;
232 SUCCEEDED(pSourceReader->GetNativeMediaType(MF_SOURCE_READER_FIRST_VIDEO_STREAM,
233 dwMediaTypeIndex, &pMediaType));
234 ++dwMediaTypeIndex
235 )
236 {
237 // スコープを抜けるときにメディアタイプを解放する
238 struct MediaTypeReleaser { IMFMediaType*& p; ~MediaTypeReleaser() { SafeRelease(&p); } } releaser{ pMediaType };
239
240 // メディアタイプを取得する
241 GUID majorType{};
242 if (FAILED(pMediaType->GetGUID(MF_MT_MAJOR_TYPE, &majorType))) continue;
243
244 // メディアタイプがビデオで無ければ次へ
245 if (majorType != MFMediaType_Video) continue;
246
247 // ビデオフォーマットを取得する
248 GUID subType{};
249 if (FAILED(pMediaType->GetGUID(MF_MT_SUBTYPE, &subType))) continue;
250
251 // 解像度を取得する
252 UINT32 width{ 0 }, height{ 0 };
253 if (FAILED(MFGetAttributeSize(pMediaType, MF_MT_FRAME_SIZE, &width, &height))) continue;
254
255 // フレームレートを取得する
256 UINT32 numerator{ 0 }, denominator{ 0 };
257 if (FAILED(MFGetAttributeRatio(pMediaType, MF_MT_FRAME_RATE, &numerator, &denominator))) continue;
258
259 // コーデックの文字列を取り出す
260 const auto& codecName{ SubTypeToName(subType) };
261
262 // 対応できないコーデックかフレームレートに問題があれば次へ
263 if (codecName.empty() || denominator == 0 || numerator == 0) continue;
264
265 // フレームレートを求める
266 const double fps{ static_cast<double>(numerator) / static_cast<double>(denominator) };
267
268 // フレームレートが 5 未満なら次へ
269 if (fps < 5.0) continue;
270
271 // UI が Media Foundation 固有の型を扱わずに済むよう、解像度と fps を表示文字列に変換する
272 std::ostringstream resolution;
273 resolution << width << " x " << height;
274 std::ostringstream frameRate;
275 frameRate << std::fixed << std::setprecision(2) << fps;
276
277 // availableFormats と同じ並び順の選択番号を関連付けて UI 用リストへ追加する
278 formatList.push_back({ resolution.str(), frameRate.str(), codecName,
279 static_cast<int>(availableFormats.size()) });
280
281 // 使用可能なビデオフォーマットのリストに追加する
282 availableFormats.emplace_back(width, height, numerator, denominator, subType);
283 }
284
285 // リストが空でなければ成功
286 return !formatList.empty();
287}

289//
290// 指定されたサブタイプに対応するビデオデコーダを探す
291//
292HRESULT CamMf::findVideoDecoder(
293 const GUID& subtype,
294 IMFTransform** ppDecoder,
295 BOOL bAllowAsync,
296 BOOL bAllowHardware,
297 BOOL bAllowTranscode
298) const
299{
300 // 結果
301 HRESULT hr{ S_OK };
302
303 // 見つかったデコーダの数
304 UINT32 count{ 0 };
305
306 // デコーダのアクティベーションオブジェクトのリスト
307 IMFActivate** ppActivate{ nullptr };
308
309 // 検索条件
310 MFT_REGISTER_TYPE_INFO inputInfo{ MFMediaType_Video, subtype };
311 MFT_REGISTER_TYPE_INFO outputInfo{ MFMediaType_Video, MFVideoFormat_NV12 };
312
313 // 列挙のフラグ
314 UINT32 unFlags
315 {
316 MFT_ENUM_FLAG_SYNCMFT
317 | MFT_ENUM_FLAG_LOCALMFT
318 | MFT_ENUM_FLAG_SORTANDFILTER
319 };
320
321 // 非同期デコーダも含めるなら
322 if (bAllowAsync) unFlags |= MFT_ENUM_FLAG_ASYNCMFT;
323
324 // ハードウェアデコーダも含めるなら
325 if (bAllowHardware) unFlags |= MFT_ENUM_FLAG_HARDWARE;
326
327 // トランスコード専用のデコーダも含めるなら
328 if (bAllowTranscode) unFlags |= MFT_ENUM_FLAG_TRANSCODE_ONLY;
329
330 // 条件に合う MFT のアクティベーションオブジェクトを列挙する
331 hr = MFTEnumEx(MFT_CATEGORY_VIDEO_DECODER,
332 unFlags, &inputInfo, &outputInfo, &ppActivate, &count);
333
334 // デコーダが見つからなかったらエラー
335 if (SUCCEEDED(hr) && count == 0) hr = MF_E_TOPO_CODEC_NOT_FOUND;
336
337 // リストの最初のデコーダをインスタンス化する
338 if (SUCCEEDED(hr)) hr = ppActivate[0]->ActivateObject(IID_PPV_ARGS(ppDecoder));
339
340 // デコーダのアクティベーションオブジェクトのリストを解放する
341 for (UINT32 i = 0; i < count; i++) ppActivate[i]->Release();
342
343 // デコーダのアクティベーションオブジェクトのリストに使ったメモリを解放する
344 CoTaskMemFree(ppActivate);
345
346 // 結果を返す
347 return hr;
348}

350//
351// MFT のセットアップと接続を行う
352//
353HRESULT CamMf::setUpPipeline(IMFTransform* pTransform, const VideoFormat& format, const GUID& subType) const
354{
355#if defined(_DEBUG)
356 std::cerr << SubTypeToName(format.subType) << " -> " << SubTypeToName(subType) << std::endl;
357#endif
358
359 // 入力と出力のメディアタイプ
360 IMFMediaType* pInputType{ nullptr };
361 IMFMediaType* pOutputType{ nullptr };
362
363 // 結果
364 HRESULT hr{ S_OK };
365
366 // MFT の入力タイプの設定
367 hr = MFCreateMediaType(&pInputType);
368 if (SUCCEEDED(hr)) hr = pInputType->SetGUID(MF_MT_MAJOR_TYPE, MFMediaType_Video);
369 if (SUCCEEDED(hr)) hr = pInputType->SetGUID(MF_MT_SUBTYPE, format.subType);
370 if (SUCCEEDED(hr)) hr = MFSetAttributeSize(pInputType, MF_MT_FRAME_SIZE, format.width, format.height);
371 if (SUCCEEDED(hr)) hr = MFSetAttributeRatio(pInputType, MF_MT_FRAME_RATE, format.fpsNum, format.fpsDenom);
372
373 // MFT に入力タイプを設定する
374 if (SUCCEEDED(hr)) hr = pTransform->SetInputType(0, pInputType, 0);
375 if (FAILED(hr)) goto done;
376
377 // MFT の出力タイプの設定
378 hr = MFCreateMediaType(&pOutputType);
379 if (SUCCEEDED(hr)) hr = pOutputType->SetGUID(MF_MT_MAJOR_TYPE, MFMediaType_Video);
380 if (SUCCEEDED(hr)) hr = pOutputType->SetGUID(MF_MT_SUBTYPE, subType);
381 if (SUCCEEDED(hr)) hr = MFSetAttributeSize(pOutputType, MF_MT_FRAME_SIZE, format.width, format.height);
382 if (SUCCEEDED(hr)) hr = MFSetAttributeRatio(pOutputType, MF_MT_FRAME_RATE, format.fpsNum, format.fpsDenom);
383
384 // MFT に出力タイプを設定する
385 if (SUCCEEDED(hr)) hr = pTransform->SetOutputType(0, pOutputType, 0);
386 if (FAILED(hr)) goto done;
387
388 // MFT をアクティブにする
389 hr = pTransform->ProcessMessage(MFT_MESSAGE_COMMAND_FLUSH, NULL);
390 if (SUCCEEDED(hr)) hr = pTransform->ProcessMessage(MFT_MESSAGE_NOTIFY_START_OF_STREAM, NULL);
391 if (SUCCEEDED(hr)) hr = pTransform->ProcessMessage(MFT_MESSAGE_NOTIFY_BEGIN_STREAMING, NULL);
392
393done:
394
395 // メディアタイプの取得に使ったメモリを解放する
396 SafeRelease(&pInputType);
397 SafeRelease(&pOutputType);
398
399 // 結果を返す
400 return hr;
401}

421//
422// デコーダの出力バッファを作成する
423//
424HRESULT CamMf::createDecoderBuffer()
425{
426 // デコーダの出力ストリーム情報を取得する
427 MFT_OUTPUT_STREAM_INFO streamInfo{ 0 };
428 if (SUCCEEDED(pDecoder->GetOutputStreamInfo(0, &streamInfo))) {}
429
430 // NV12 は通常 16 ピクセル単位に切り上げられたサイズなので幅を 16 の倍数に丸める
431 const auto alignedWidth{ static_cast<UINT32>((width + 15) & ~15) };
432 const auto alignedHeight{ static_cast<UINT32>((height + 15) & ~15) };
433
434 // NV12 フォーマットの理論上の必要バッファサイズを計算する
435 const auto cbDecoderCalc{ static_cast<UINT32>((alignedWidth * alignedHeight * 3) / 2) };
436
437 // MFT が要求するサイズと理論計算値の大きい方を採用する
438 const auto cbDecoder{ static_cast<UINT32>((streamInfo.cbSize > cbDecoderCalc) ? streamInfo.cbSize : cbDecoderCalc) };
439
440 // MFT が要求するメモリアラインメントを取得する
441 const auto alignmentDecoder{ static_cast<DWORD>((streamInfo.cbAlignment > 0) ? (streamInfo.cbAlignment - 1) : 63) };
442
443 // 現在の出力フォーマット用に確保されている出力バッファがあれば解放する
444 SafeRelease(&pDecoderBuffer);
445
446 // デコーダの出力フレームを書き込むためのアラインメント付きメモリバッファを作成する
447 return MFCreateAlignedMemoryBuffer(cbDecoder, alignmentDecoder, &pDecoderBuffer);
448}

450//
451// カラーコンバータの出力バッファを作成する
452//
453HRESULT CamMf::createConverterBuffer()
454{
455 // カラーコンバータの出力ストリーム情報を取得する
456 MFT_OUTPUT_STREAM_INFO convStreamInfo{ 0 };
457 if (SUCCEEDED(pConverter->GetOutputStreamInfo(0, &convStreamInfo))) {}
458
459 // RGB32 フォーマットの理論上の必要バッファサイズを計算する
460 UINT32 cbConverterCalc{ 0 };
461 MFCalculateImageSize(MFVideoFormat_RGB32, width, height, &cbConverterCalc);
462
463 // MFT が要求するサイズと理論計算値の大きい方を採用する
464 UINT32 cbConverter = (convStreamInfo.cbSize > cbConverterCalc) ? convStreamInfo.cbSize : cbConverterCalc;
465
466 // MFT が要求するメモリアラインメントを取得する
467 DWORD alignmentConverter = (convStreamInfo.cbAlignment > 0) ? (convStreamInfo.cbAlignment - 1) : 63;
468
469 // 既存の出力バッファがあれば解放する
470 SafeRelease(&pConverterBuffer);
471
472 // カラーコンバータが要求するサイズとアラインメントを満たす RGB32 用の出力バッファを作成する
473 return MFCreateAlignedMemoryBuffer(cbConverter, alignmentConverter, &pConverterBuffer);
474}

476//
477// Source Reader の出力フォーマットを設定し、基底クラスの frame を初期化する
478//
479bool CamMf::setFormat(int index)
480{
481 // 使用可能なフォーマットのリストから選択されたフォーマットを取得する
482 VideoFormat selectedFormat{ availableFormats[index] };
483
484 // デコード方法
485 enum class DecodeMethod { None = 0, Convert, Decode } decodeMethod
486 {
487 selectedFormat.subType == MFVideoFormat_MJPG ? DecodeMethod::Decode :
488 selectedFormat.subType == MFVideoFormat_H264 ? DecodeMethod::Decode :
489 selectedFormat.subType == MFVideoFormat_NV12 ? DecodeMethod::Convert :
490 selectedFormat.subType == MFVideoFormat_YUY2 ? DecodeMethod::Convert :
491 DecodeMethod::None
492 };
493
494 // デコーダとカラーコンバータを未設定にする
495 cleanUpTransform(&pDecoder);
496 cleanUpTransform(&pConverter);
497
498 // 結果
499 HRESULT hr{ S_OK };
500
501 // 新しいメディアタイプを作成する
502 IMFMediaType* pMediaType{ nullptr };
503 hr = MFCreateMediaType(&pMediaType);
504 if (FAILED(hr)) goto done;
505
506 // メジャータイプにビデオを指定する
507 hr = pMediaType->SetGUID(MF_MT_MAJOR_TYPE, MFMediaType_Video);
508 if (FAILED(hr)) goto done;
509
510 // ビデオフォーマットを指定する
511 hr = pMediaType->SetGUID(MF_MT_SUBTYPE, selectedFormat.subType);
512 if (FAILED(hr)) goto done;
513
514 // 解像度を設定する
515 hr = MFSetAttributeSize(pMediaType, MF_MT_FRAME_SIZE, selectedFormat.width, selectedFormat.height);
516 if (FAILED(hr)) goto done;
517
518 // フレームレートを設定する
519 hr = MFSetAttributeRatio(pMediaType, MF_MT_FRAME_RATE, selectedFormat.fpsNum, selectedFormat.fpsDenom);
520 if (FAILED(hr)) goto done;
521
522 // Source Reader の出力メディアタイプを設定する
523 hr = pSourceReader->SetCurrentMediaType(MF_SOURCE_READER_FIRST_VIDEO_STREAM, nullptr, pMediaType);
524 if (FAILED(hr)) goto done;
525
526 // 基底クラスの解像度とチャンネル数を設定し、バッファサイズを設定する
527 width = selectedFormat.width;
528 height = selectedFormat.height;
529 channels = 4;
530 frame.resize(width * height * channels);
531 image.resize(width * height * channels);
532
533 // インターバルを計算する
534 interval = (selectedFormat.fpsDenom != 0)
535 ? (1000.0 * selectedFormat.fpsDenom / selectedFormat.fpsNum)
536 : 10.0;
537
538 // デコードまたはカラー変換が必要なとき
539 if (decodeMethod != DecodeMethod::None)
540 {
541 // デコードが必要なら
542 if (decodeMethod == DecodeMethod::Decode)
543 {
544 // 選択されたビデオフォーマットのデコーダを探す
545 hr = findVideoDecoder(selectedFormat.subType, &pDecoder);
546 if (FAILED(hr)) goto done;
547
548 // デコーダから ICodecAPI インターフェイスの取得に成功したら
549 ICodecAPI* pCodecAPI{ nullptr };
550 if (SUCCEEDED(pDecoder->QueryInterface(IID_PPV_ARGS(&pCodecAPI))))
551 {
552 // コーデック設定値を渡すための VARIANT を初期化する
553 VARIANT var;
554 VariantInit(&var);
555
556 // Low Latency Mode を VT_UI4 型で有効にする
557 var.vt = VT_UI4;
558
559 // 1 = Low Latency Mode
560 var.ulVal = 1;
561
562 // デコーダに Low Latency Mode を要求する
563 hr = pCodecAPI->SetValue(&CODECAPI_AVLowLatencyMode, &var);
564
565 // VT_UI4 型で Low Latency Mode を有効にできなかったら
566 if (FAILED(hr))
567 {
568 // Low Latency Mode を VT_BOOL 型で有効にする
569 var.vt = VT_BOOL;
570
571 // VARIANT_TRUE = Low Latency Mode
572 var.boolVal = VARIANT_TRUE;
573
574 // デコーダに Low Latency Mode を要求する
575 pCodecAPI->SetValue(&CODECAPI_AVLowLatencyMode, &var);
576 }
577
578 // VARIANT が保持しているリソースを解放する38
579 VariantClear(&var);
580
581 // ICodecAPI インターフェイスを解放する
582 SafeRelease(&pCodecAPI);
583 }
584
585 // MFT デコーダのセットアップと接続を行う
586 hr = setUpPipeline(pDecoder, selectedFormat, MFVideoFormat_NV12);
587 if (FAILED(hr)) goto done;
588
589 // デコード後のビデオフォーマットは NV12 にしている
590 selectedFormat.subType = MFVideoFormat_NV12;
591
592 // デコーダの出力バッファを作成する
593 hr = createDecoderBuffer();
594 if (FAILED(hr)) goto done;
595 }
596
597 // MFT をインスタンス化する
598 hr = CoCreateInstance(CLSID_CColorConvertDMO, nullptr, CLSCTX_INPROC_SERVER, IID_PPV_ARGS(&pConverter));
599 if (FAILED(hr)) goto done;
600

602 hr = setUpPipeline(pConverter, selectedFormat, MFVideoFormat_RGB32);
603 if (FAILED(hr)) goto done;
604
605 // カラー変換用の出力バッファを作成する
606 hr = createConverterBuffer();
607 if (FAILED(hr)) goto done;
608 }
609
610done:
611
612 // メディアタイプはもう使わないので解放する
613 SafeRelease(&pMediaType);
614
615 // 成功したら true を返す
616 if (SUCCEEDED(hr)) return true;
617
618 // 失敗したときはカメラを閉じて
619 close();
620
621 // false を返す
622 return false;

692//
693// フレームをキャプチャする
694//
695void CamMf::capture()
696{
697 // スレッドが実行可の間
698 while (running)
699 {
700 // サンプルを読み出して
701 DWORD dwStreamIndex{ 0 };
702 DWORD dwStreamFlags{ 0 };
703 LONGLONG llTimestamp{ 0 };
704 IMFSample* pSample{ nullptr };
705 if (FAILED(pSourceReader->ReadSample(MF_SOURCE_READER_FIRST_VIDEO_STREAM,
706 0, &dwStreamIndex, &dwStreamFlags, &llTimestamp, &pSample))) continue;
707
708 // サンプルが取得できていなければ戻る
709 if (!pSample) continue;
710
711 // サンプルから取り出したメディアバッファ
712 IMFMediaBuffer* pBuffer{ nullptr };
713
714 // デコーダの出力フレームを保持するサンプル
715 IMFSample* pDecodedSample{ nullptr };
716
717 // カラーコンバータの出力フレームを保持するサンプル
718 IMFSample* pConvertedSample{ nullptr };
719
720 // ProcessOutput の呼び出しの状態
721 DWORD dwStatus{ 0 };
722
723 // デコーダが設定されていれば (MJPG か H264 の場合)
724 if (pDecoder)
725 {
726 // サンプルをデコーダに渡す
727 HRESULT hr{ pDecoder->ProcessInput(0, pSample, 0) };
728
729 // デコーダへの入力に失敗したら
730 if (FAILED(hr))
731 {
732#if defined(_DEBUG)
733 if (hr != MF_E_NOTACCEPTING)
734 {
735 std::cerr << "Decoder process input failed: ";
736 switch (hr)
737 {
738 case E_INVALIDARG:
739 std::cerr << "Invalid argument." << std::endl; break;
740 case E_UNEXPECTED:
741 std::cerr << "Unexpected error." << std::endl; break;
742 case E_FAIL:
743 std::cerr << "Unspecified error." << std::endl; break;
744 case MF_E_INVALIDSTREAMNUMBER:
745 std::cerr << "Invalid stream number." << std::endl; break;
746 case MF_E_NO_SAMPLE_DURATION:
747 std::cerr << "No sample duration." << std::endl; break;
748 case MF_E_NO_SAMPLE_TIMESTAMP:
749 std::cerr << "No sample timestamp." << std::endl; break;
750 case MF_E_NOTACCEPTING:
751 std::cerr << "Not accepting input." << std::endl; break;
752 case MF_E_TRANSFORM_TYPE_NOT_SET:
753 std::cerr << "Transform type not set." << std::endl; break;
754 case MF_E_UNSUPPORTED_D3D_TYPE:
755 std::cerr << "Unsupported D3D type." << std::endl; break;
756 default:
757 std::cerr << "Code: " << std::hex << hr << std::endl; break;
758 }
759 }
760#endif
761 goto done;
762 }
763
764 // デコーダの出力ストリーム情報を取得し、誰がサンプルを提供するべきかを判定する
765 MFT_OUTPUT_STREAM_INFO streamInfo{};
766 bool mftProvidesSamples{ false };
767 if (SUCCEEDED(pDecoder->GetOutputStreamInfo(0, &streamInfo)))
768 {
769 mftProvidesSamples = (streamInfo.dwFlags & MFT_OUTPUT_STREAM_PROVIDES_SAMPLES) != 0;
770 }
771
772 // デコーダから出力サンプルを取り出す
773 IMFSample* pLatestDecodedSample{ nullptr };
774 bool hasOutput{ false };
775
776 while (true)
777 {
778 DWORD prevLength = 0;
779
780 if (mftProvidesSamples)
781 {
782 // MFT が自前でサンプルを割り当てるので nullptr を渡す
783 pDecodedSample = nullptr;
784 }
785 else
786 {
787 // 呼び出し側がサンプルを提供する
788 hr = MFCreateSample(&pDecodedSample);
789 if (FAILED(hr)) goto done;
790
791 // クラスメンバの pDecoderBuffer が有効であることを確認
792 if (!pDecoderBuffer)
793 {
794 hr = createDecoderBuffer();
795 if (FAILED(hr))
796 {
797 SafeRelease(&pDecodedSample);
798 goto done;
799 }
800 }
801
802 // バッファの有効データ長を 0 に設定して、MFT が先頭から書き込めるようにする
803 pDecoderBuffer->GetCurrentLength(&prevLength);
804 pDecoderBuffer->SetCurrentLength(0);
805
806 hr = pDecodedSample->AddBuffer(pDecoderBuffer);
807 if (FAILED(hr))
808 {
809 SafeRelease(&pDecodedSample);
810 goto done;
811 }
812 }
813
814 MFT_OUTPUT_DATA_BUFFER decodedBuffer{ 0, pDecodedSample, 0, nullptr };
815 hr = pDecoder->ProcessOutput(0, 1, &decodedBuffer, &dwStatus);
816
817 // ストリームのフォーマットが変化した場合の処理
818 if (hr == MF_E_TRANSFORM_STREAM_CHANGE)
819 {
820#if defined(_DEBUG)
821 std::cerr << "Transform stream change." << std::endl;
822#endif
823 // 1. 利用可能な出力タイプから NV12 を探索・選択する
824 IMFMediaType* pNewOutputType{ nullptr };
825 GUID subtype{ 0 };
826 DWORD dwTypeIndex = 0;
827 bool foundNV12 = false;
828 hr = S_OK; // hr を S_OK にリセット
829 while (SUCCEEDED(pDecoder->GetOutputAvailableType(0, dwTypeIndex++, &pNewOutputType)))
830 {
831 GUID subType{};
832 if (SUCCEEDED(pNewOutputType->GetGUID(MF_MT_SUBTYPE, &subType)) && subType == MFVideoFormat_NV12)
833 {
834 foundNV12 = true;
835 subtype = subType;
836#if defined(_DEBUG)
837 std::cerr << "Found NV12 output type at index " << (dwTypeIndex - 1) << std::endl;
838#endif
839 break;
840 }
841 SafeRelease(&pNewOutputType);
842 }
843
844 // 見つからなかった場合はインデックス 0 を取得してサブタイプを NV12 に上書きする
845 if (!foundNV12)
846 {
847#if defined(_DEBUG)
848 std::cerr << "NV12 output type not found. Forcing fallback to index 0." << std::endl;
849#endif
850 hr = pDecoder->GetOutputAvailableType(0, 0, &pNewOutputType);
851 if (SUCCEEDED(hr))
852 {
853 hr = pNewOutputType->SetGUID(MF_MT_SUBTYPE, MFVideoFormat_NV12);
854 subtype = MFVideoFormat_NV12;
855 }
856 }
857
858 // 2. 現在の入力メディアタイプから解像度などの属性をコピーする
859 if (SUCCEEDED(hr) && pNewOutputType)
860 {
861 IMFMediaType* pInputType{ nullptr };
862 if (SUCCEEDED(pDecoder->GetInputCurrentType(0, &pInputType)))
863 {
864 UINT32 w{ 0 }, h{ 0 };
865 if (SUCCEEDED(MFGetAttributeSize(pInputType, MF_MT_FRAME_SIZE, &w, &h)))
866 {
867 MFSetAttributeSize(pNewOutputType, MF_MT_FRAME_SIZE, w, h);
868 }
869
870 UINT32 fpsNum{ 0 }, fpsDenom{ 0 };
871 if (SUCCEEDED(MFGetAttributeRatio(pInputType, MF_MT_FRAME_RATE, &fpsNum, &fpsDenom)))
872 {
873 MFSetAttributeRatio(pNewOutputType, MF_MT_FRAME_RATE, fpsNum, fpsDenom);
874 }
875
876 UINT32 aspectNum{ 0 }, aspectDenom{ 0 };
877 if (SUCCEEDED(MFGetAttributeRatio(pInputType, MF_MT_PIXEL_ASPECT_RATIO, &aspectNum, &aspectDenom)))
878 {
879 MFSetAttributeRatio(pNewOutputType, MF_MT_PIXEL_ASPECT_RATIO, aspectNum, aspectDenom);
880 }
881
882 PROPVARIANT var;
883 PropVariantInit(&var);
884 if (SUCCEEDED(pInputType->GetItem(MF_MT_INTERLACE_MODE, &var)))
885 {
886 if (var.vt == VT_UI4) pNewOutputType->SetUINT32(MF_MT_INTERLACE_MODE, var.ulVal);
887 PropVariantClear(&var);
888 }
889
890 SafeRelease(&pInputType);
891 }
892
893 // 新しい出力メディアタイプを設定する
894 hr = pDecoder->SetOutputType(0, pNewOutputType, 0);
895#if defined(_DEBUG)
896 if (FAILED(hr)) std::cerr << "pDecoder->SetOutputType failed: " << std::hex << hr << std::endl;
897 else std::cerr << "pDecoder->SetOutputType succeeded." << std::endl;
898#endif
899 }
900 else if (SUCCEEDED(hr))
901 {
902 hr = E_FAIL;
903#if defined(_DEBUG)
904 std::cerr << "No output type available to set!" << std::endl;
905#endif
906 }
907
908 // 3. 設定された出力タイプから属性を取得してバッファとカラーコンバータを更新する
909 if (SUCCEEDED(hr) && pNewOutputType)
910 {
911 UINT32 newWidth{ 0 }, newHeight{ 0 };
912 if (SUCCEEDED(MFGetAttributeSize(pNewOutputType, MF_MT_FRAME_SIZE, &newWidth, &newHeight)) && newWidth > 0 && newHeight > 0)
913 {
914 width = newWidth;
915 height = newHeight;
916 }
917
918 UINT32 fpsNum{ 0 }, fpsDenom{ 0 };
919 MFGetAttributeRatio(pNewOutputType, MF_MT_FRAME_RATE, &fpsNum, &fpsDenom);
920
921 // デコーダの出力バッファ要件の取得と再作成
922 hr = createDecoderBuffer();
923
924 // カラーコンバータの再設定
925 if (SUCCEEDED(hr) && pConverter)
926 {
927 VideoFormat newFormat{ static_cast<UINT32>(width), static_cast<UINT32>(height), fpsNum, fpsDenom, subtype };
928
929 // カラーコンバータのセットアップ
930 hr = setUpPipeline(pConverter, newFormat, MFVideoFormat_RGB32);
931
932 if (SUCCEEDED(hr))
933 {
934 hr = createConverterBuffer();
935 }
936 }
937
938 // 基底クラスのバッファリサイズ
939 if (SUCCEEDED(hr))
940 {
941 frame.resize(width * height * channels);
942 image.resize(width * height * channels);
943 }
944 }
945
946 // 出力サンプルとイベントを破棄する
947 if (!mftProvidesSamples) SafeRelease(&pDecodedSample);
948 else SafeRelease(&decodedBuffer.pSample);
949 SafeRelease(&decodedBuffer.pEvents);
950
951 // リソースの解放
952 SafeRelease(&pNewOutputType);
953
954 if (FAILED(hr))
955 {
956#if defined(_DEBUG)
957 std::cerr << "Failed to handle stream change: " << std::hex << hr << std::endl;
958#endif
959 goto done;
960 }
961
962 // ストリーム情報を再取得する（ストリームチェンジによってフラグが変わる可能性があるため）
963 if (SUCCEEDED(pDecoder->GetOutputStreamInfo(0, &streamInfo)))
964 {
965 mftProvidesSamples = (streamInfo.dwFlags & MFT_OUTPUT_STREAM_PROVIDES_SAMPLES) != 0;
966#if defined(_DEBUG)
967 std::cerr << "After stream change, mftProvidesSamples = " << (mftProvidesSamples ? "true" : "false") << ", cbSize = " << streamInfo.cbSize << std::endl;
968#endif
969 }
970
971#if defined(_DEBUG)
972 std::cerr << "Stream change handled successfully, retrying ProcessOutput." << std::endl;
973#endif
974 // ストリーム変更を行ったので、今回の ProcessOutput はやり直す
975 if (!mftProvidesSamples && pDecoderBuffer)
976 {
977 pDecoderBuffer->SetCurrentLength(prevLength);
978 }
979 continue;
980 }
981
982 // デコーダがさらなる入力を要求している場合 (正常動作)
983 if (hr == MF_E_TRANSFORM_NEED_MORE_INPUT)
984 {
985 if (!mftProvidesSamples)
986 {
987 if (pDecoderBuffer) pDecoderBuffer->SetCurrentLength(prevLength);
988 SafeRelease(&pDecodedSample);
989 }
990 else
991 {
992 SafeRelease(&decodedBuffer.pSample);
993 }
994 break;
995 }
996
997 // その他のエラー
998 if (FAILED(hr))
999 {
1000#if defined(_DEBUG)
1001 std::cerr << "Decoder process output failed (loop bottom): " << std::hex << hr << std::endl;
1002#endif
1003 if (!mftProvidesSamples)
1004 {
1005 if (pDecoderBuffer) pDecoderBuffer->SetCurrentLength(prevLength);
1006 SafeRelease(&pDecodedSample);
1007 }
1008 else
1009 {
1010 SafeRelease(&decodedBuffer.pSample);
1011 }
1012 SafeRelease(&decodedBuffer.pEvents);
1013 goto done;
1014 }
1015
1016 // デコード成功！
1017 SafeRelease(&pLatestDecodedSample);
1018 if (mftProvidesSamples)
1019 {
1020 pLatestDecodedSample = decodedBuffer.pSample;
1021 }
1022 else
1023 {
1024 pLatestDecodedSample = pDecodedSample;
1025 }
1026 pDecodedSample = nullptr;
1027 SafeRelease(&decodedBuffer.pEvents);
1028 hasOutput = true;
1029
1030 // レイテンシ優先なら
1031 if (prioritizeLatency)
1032 {
1033 // 最新を取得し続けるためループを継続する
1034 continue;
1035 }
1036 else
1037 {
1038 // 全フレーム処理モードならループを抜ける
1039 break;
1040 }
1041 }
1042
1043 if (hasOutput && pLatestDecodedSample)
1044 {
1045 // 元のサンプルはもう使わないので解放する
1046 pSample->Release();
1047
1048 // デコードに成功したらデコードされたサンプルを使う
1049 pSample = pLatestDecodedSample;
1050 pLatestDecodedSample = nullptr;
1051 }
1052 else
1053 {
1054 // 今回の入力ではデコード完了フレームが得られなかったため、
1055 // 入力サンプルを解放して次のフレームの読み込みに進む
1056 pSample->Release();
1057 pSample = nullptr;
1058 continue;
1059 }
1060 }
1061
1062 // カラーコンバータが設定されていれば (NV12 か YUY2 か MJPG が H264 の場合)
1063 if (pConverter)
1064 {
1065 // サンプルをカラーコンバータに渡す
1066 HRESULT hr{ pConverter->ProcessInput(0, pSample, 0) };
1067 if (FAILED(hr)) goto done;
1068
1069 // カラーコンバータの出力サンプルを作成する
1070 hr = MFCreateSample(&pConvertedSample);
1071 if (FAILED(hr)) goto done;
1072
1073 // バッファの長さを 0 に設定して書き込み可能にする
1074 pConverterBuffer->SetCurrentLength(0);
1075
1076 // カラーコンバータの出力サンプルにバッファを追加する
1077 hr = pConvertedSample->AddBuffer(pConverterBuffer);
1078 if (FAILED(hr)) goto done;
1079
1080 // カラーコンバータの出力バッファ
1081 MFT_OUTPUT_DATA_BUFFER convertedBuffer{ 0, pConvertedSample, 0, nullptr };
1082
1083 // カラー変換処理を実行
1084 hr = pConverter->ProcessOutput(0, 1, &convertedBuffer, &dwStatus);
1085 if (FAILED(hr)) goto done;
1086
1087 // 元のサンプルはもう使わないので解放する
1088 pSample->Release();
1089
1090 // コンバートに成功したらコンバートされたサンプルを使う
1091 pSample = pConvertedSample;
1092 pConvertedSample = nullptr;
1093 }
1094
1095 // サンプルからメディアバッファを取得して
1096 if (FAILED(pSample->GetBufferByIndex(0, &pBuffer))) goto done;
1097
1098 // メディアバッファが取得できていれば
1099 if (pBuffer)
1100 {
1101 // メディアバッファから取り出したデータ
1102 BYTE* pData{ nullptr };
1103
1104 // メディアバッファから取り出したデータの長さ
1105 DWORD cbDataLength{ 0 };
1106
1107 // メディアバッファからフレームの情報を取得できたら
1108 if (SUCCEEDED(pBuffer->Lock(&pData, nullptr, &cbDataLength)) && pData)
1109 {
1110 // 全フレーム処理モードなら
1111 if (!prioritizeLatency)
1112 {
1113 // キャプチャしている間は
1114 while (running && captured)
1115 {
1116 // スレッドを一時停止して CPU リソースを解放する
1117 std::this_thread::yield();
1118 }
1119 }
1120
1121 // 一時メモリをロックして
1122 std::lock_guard<std::mutex> lock{ mtx };
1123
1124 // キャプチャしたデータを一時メモリにコピーしたら
1125 if (image.size() < cbDataLength) image.resize(cbDataLength);
1126 memcpy(image.data(), pData, cbDataLength);
1127
1128 // 新しいフレームがキャプチャされたことを通知して
1129 captured = true;
1130
1131 // メディアバッファのロックを解除する
1132 pBuffer->Unlock();
1133 }
1134
1135 // メディアバッファを解放する
1136 pBuffer->Release();
1137 pBuffer = nullptr;
1138 }
1139
1140 done:
1141
1142 // デコーダの出力サンプルとバッファを解放する
1143 if (pDecodedSample) pDecodedSample->Release();
1144
1145 // カラーコンバータの出力サンプルバッファを解放する
1146 if (pConvertedSample) pConvertedSample->Release();
1147
1148 // サンプルを解放する
1149 pSample->Release();
1150 pSample = nullptr;
1151 }
1152}
