189 bool transmit(GLuint buffer)
190 {
191 // 新しいフレームが取得されているときカメラのロックが成功したら
192 std::unique_lock<std::mutex> lock(mtx, std::try_to_lock);
193 if (lock.owns_lock() && captured)
194 {
195 // データの長さを計算して
196 const auto expected_length{ static_cast<size_t>(width) * height * channels };
197 const auto length{ std::min(image.size(), expected_length) * sizeof(GLubyte) };
198
199 // フレームをピクセルバッファオブジェクトに転送して
200 glBindBuffer(GL_PIXEL_PACK_BUFFER, buffer);
201 glBufferSubData(GL_PIXEL_PACK_BUFFER, 0, length, image.data());
202 glBindBuffer(GL_PIXEL_PACK_BUFFER, 0);
203
204 // 動画・カメラ入力では次のフレームを待つ。静止画像は表示方式を
205 // 切り替えた後も同じ内容を再転送できるよう取得済みの状態を維持する。
206 if (!reusableFrame) captured = false;
207 return true;
208 }
209
210 // カメラがロックできなかった
211 return false;
212 }
