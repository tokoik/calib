33class GgOpenXR::Impl
34{
35public:
36
37 // OpenXR 型を含まない公開用データは
38 // OpenXR 無効ビルドでも同じ API を提供できるよう
39 // 条件付きコンパイルの外に置く。
40
41 // PRIMARY_STEREO 構成で列挙された view の数.
42 std::vector<View> publicViews;
43
44 // 基準空間における HMD 中央の姿勢.
45 Pose headPose;
46
47 // HMD 中央の姿勢が有効かどうか.
48 bool poseValid{ false };
49
50 // ランタイムからアプリケーションの終了が要求されているかどうか.
51 bool closeRequested{ false };
52
53 // 現在フレームを描画するようランタイムが要求しているかどうか.
54 bool renderRequested{ false };
55
56#if defined( GG_ENABLE_OPENXR)
57 // 一つの view に対応する swapchain と OpenGL 側の描画資源。
58 // images はランタイム所有のテクスチャであり、fbos だけを本クラスが所有する。

60 // OpenXR の swapchain image。
61 // ランタイムが所有するため、
62 // OpenGL 側で FBO を作成して描画する。
63 struct Swapchain
64 {
65 XrSwapchain handle{ XR_NULL_HANDLE };
66 uint32_t width{ 0 };
67 uint32_t height{ 0 };
68 uint32_t imageIndex{ 0 };
69 bool acquired{ false };
70 std::vector<XrSwapchainImageOpenGLKHR> images;
71 std::vector<GLuint> fbos;
72 };
73
74 /* OpenXR 資源は instance -> session -> space/swapchain の親子関係を持つ。 */
75 /* destroy() では必ずこの逆順で破棄する。 */
76 XrInstance instance{ XR_NULL_HANDLE };
77 XrSystemId systemId{ XR_NULL_SYSTEM_ID };
78 XrSession session{ XR_NULL_HANDLE };
79 XrSpace space{ XR_NULL_HANDLE };
80 XrSpace viewSpace{ XR_NULL_HANDLE };
81 XrSessionState sessionState{ XR_SESSION_STATE_UNKNOWN };
82 XrFrameState frameState{ XR_TYPE_FRAME_STATE };
83 bool sessionRunning{ false };
84 bool frameActive{ false };
85 std::vector<XrView> views;

93 void destroy()
94 {
95 /* beginFrame() 後に例外的な終了経路へ入っても、OpenXR の呼び出し順序を */
96 /* 壊さないよう空の xrEndFrame() で未完了フレームを閉じる。 */
97 if (frameActive && session != XR_NULL_HANDLE)
98 {
99 XrFrameEndInfo endInfo{ XR_TYPE_FRAME_END_INFO };
100 endInfo.displayTime = frameState.predictedDisplayTime;
101 endInfo.environmentBlendMode = XR_ENVIRONMENT_BLEND_MODE_OPAQUE;
102 xrEndFrame(session, &endInfo);
103 frameActive = false;
104 }
105
106 for (auto& swapchain : swapchains)
107 {
108 /* acquire 済み画像はランタイムへ返してから swapchain を破棄する。 */
109 if (swapchain.acquired)
110 {
111 XrSwapchainImageReleaseInfo releaseInfo{ XR_TYPE_SWAPCHAIN_IMAGE_RELEASE_INFO };
112 xrReleaseSwapchainImage(swapchain.handle, &releaseInfo);
113 }
114 if (!swapchain.fbos.empty())
115 glDeleteFramebuffers(static_cast<GLsizei>(swapchain.fbos.size()), swapchain.fbos.data());
116 if (swapchain.handle != XR_NULL_HANDLE) xrDestroySwapchain(swapchain.handle);
117 }
118 swapchains.clear();
119 views.clear();
120 publicViews.clear();
121
122 /* 子オブジェクトから親オブジェクトの順に解放する。 */
123 if (viewSpace != XR_NULL_HANDLE) xrDestroySpace(viewSpace);
124 viewSpace = XR_NULL_HANDLE;
125 if (space != XR_NULL_HANDLE) xrDestroySpace(space);
126 space = XR_NULL_HANDLE;
127 if (sessionRunning && session != XR_NULL_HANDLE) xrEndSession(session);
128 sessionRunning = false;
129 if (session != XR_NULL_HANDLE) xrDestroySession(session);
130 session = XR_NULL_HANDLE;
131 if (instance != XR_NULL_HANDLE) xrDestroyInstance(instance);
132 instance = XR_NULL_HANDLE;
133 }
134#endif
135};

146bool GgOpenXR::initialize(GLFWwindow* window, const std::string& applicationName)
147{
148#if !defined( GG_ENABLE_OPENXR)
149 (void)window;
150 (void)applicationName;
151 return false;
152#else
153 /* 再初期化時にも古い session や FBO を残さない。 */
154 shutdown();
155
156 /* OpenGL の texture を swapchain image として受け取るために必要な拡張を有効化する。 */
157 const char* extensions[]{ XR_KHR_OPENGL_ENABLE_EXTENSION_NAME };
158 XrInstanceCreateInfo instanceInfo{ XR_TYPE_INSTANCE_CREATE_INFO };
159 std::strncpy(instanceInfo.applicationInfo.applicationName, applicationName.c_str(),
160 XR_MAX_APPLICATION_NAME_SIZE - 1);
161 std::strncpy(instanceInfo.applicationInfo.engineName, "GgApp", XR_MAX_ENGINE_NAME_SIZE - 1);
162 instanceInfo.applicationInfo.apiVersion = XR_CURRENT_API_VERSION;
163 instanceInfo.enabledExtensionCount = 1;
164 instanceInfo.enabledExtensionNames = extensions;
165 if (!impl->check(xrCreateInstance(&instanceInfo, &impl->instance))) return false;
166
167 /* 現在アクティブなランタイムから HMD 用 system を取得する。 */
168 XrSystemGetInfo systemInfo{ XR_TYPE_SYSTEM_GET_INFO };
169 systemInfo.formFactor = XR_FORM_FACTOR_HEAD_MOUNTED_DISPLAY;
170 if (!impl->check(xrGetSystem(impl->instance, &systemInfo, &impl->systemId)))
171 {
172 shutdown();
173 return false;
174 }
175
176 /* OpenGL graphics requirements は拡張関数なので、instance から関数ポインタを取得する。 */
177 PFN_xrGetOpenGLGraphicsRequirementsKHR getRequirements{};
178 if (!impl->check(xrGetInstanceProcAddr(impl->instance, "xrGetOpenGLGraphicsRequirementsKHR",
179 reinterpret_cast<PFN_xrVoidFunction*>(&getRequirements))))
180 {
181 shutdown();
182 return false;
183 }
184 XrGraphicsRequirementsOpenGLKHR requirements{ XR_TYPE_GRAPHICS_REQUIREMENTS_OPENGL_KHR };
185 if (!impl->check(getRequirements(impl->instance, impl->systemId, &requirements)))
186 {
187 shutdown();
188 return false;
189 }
190
191 /* OpenXR session を現在の GLFW/WGL コンテキストへ結び付ける。 */
192 /* GetDC() で借用した HDC は xrCreateSession() の直後に返却する。 */
193 const HWND hwnd{ glfwGetWin32Window(window) };
194 XrGraphicsBindingOpenGLWin32KHR binding{ XR_TYPE_GRAPHICS_BINDING_OPENGL_WIN32_KHR };
195 binding.hDC = GetDC(hwnd);
196 binding.hGLRC = wglGetCurrentContext();
197 if (!binding.hDC || !binding.hGLRC)
198 {
199 if (binding.hDC) ReleaseDC(hwnd, binding.hDC);
200 shutdown();
201 return false;
202 }
203 XrSessionCreateInfo sessionInfo{ XR_TYPE_SESSION_CREATE_INFO };
204 sessionInfo.next = &binding;
205 sessionInfo.systemId = impl->systemId;
206 const XrResult sessionResult{ xrCreateSession(impl->instance, &sessionInfo, &impl->session) };
207 ReleaseDC(hwnd, binding.hDC);
208 if (!impl->check(sessionResult))
209 {
210 shutdown();
211 return false;
212 }
213
214 /* アプリケーションの基準空間には床基準の STAGE を優先し、利用できない */
215 /* ランタイムでは起動位置基準の LOCAL へフォールバックする。 */
216 XrReferenceSpaceCreateInfo spaceInfo{ XR_TYPE_REFERENCE_SPACE_CREATE_INFO };
217 spaceInfo.referenceSpaceType = XR_REFERENCE_SPACE_TYPE_STAGE;
218 spaceInfo.poseInReferenceSpace.orientation.w = 1.0f;
219 if (XR_FAILED(xrCreateReferenceSpace(impl->session, &spaceInfo, &impl->space)))
220 {
221 spaceInfo.referenceSpaceType = XR_REFERENCE_SPACE_TYPE_LOCAL;
222 if (!impl->check(xrCreateReferenceSpace(impl->session, &spaceInfo, &impl->space)))
223 {
224 shutdown();
225 return false;
226 }
227 }
228
229 /* HMD 中央姿勢を眼ごとの pose から推測せず取得するため、VIEW 空間も作成する。 */
230 spaceInfo.referenceSpaceType = XR_REFERENCE_SPACE_TYPE_VIEW;
231 if (!impl->check(xrCreateReferenceSpace(impl->session, &spaceInfo, &impl->viewSpace)))
232 {
233 shutdown();
234 return false;
235 }
236
237 /* PRIMARY_STEREO の view 数と、各 view の推奨解像度をランタイムから取得する。 */
238 /* view 数を 2 に固定しないことで、公開 API と資源管理を列挙結果に一致させる。 */
239 uint32_t viewCount{};
240 xrEnumerateViewConfigurationViews(impl->instance, impl->systemId,
241 XR_VIEW_CONFIGURATION_TYPE_PRIMARY_STEREO, 0, &viewCount, nullptr);
242 std::vector<XrViewConfigurationView> configurations(viewCount,
243 { XR_TYPE_VIEW_CONFIGURATION_VIEW });
244 if (!impl->check(xrEnumerateViewConfigurationViews(impl->instance, impl->systemId,
245 XR_VIEW_CONFIGURATION_TYPE_PRIMARY_STEREO, viewCount, &viewCount, configurations.data())))
246 {
247 shutdown();
248 return false;
249 }
250
251 /* ランタイムが受け付ける swapchain format の中から、通常の OpenGL 描画と */
252 /* 相互運用できる sRGB RGBA、RGBA の順で選択する。 */
253 uint32_t formatCount{};
254 xrEnumerateSwapchainFormats(impl->session, 0, &formatCount, nullptr);
255 std::vector<int64_t> formats(formatCount);
256 xrEnumerateSwapchainFormats(impl->session, formatCount, &formatCount, formats.data());
257 const int64_t preferred[]{ GL_SRGB8_ALPHA8, GL_RGBA8 };
258 int64_t format{};
259 for (const auto candidate : preferred)
260 if (std::find(formats.begin(), formats.end(), candidate) != formats.end()) { format = candidate; break; }
261 if (!format)
262 {
263 shutdown();
264 return false;
265 }
266
267 impl->views.assign(viewCount, { XR_TYPE_VIEW });
268 impl->publicViews.resize(viewCount);
269 impl->swapchains.resize(viewCount);
270 for (uint32_t i{}; i < viewCount; ++i)
271 {
272 /* view ごとに独立した swapchain を作り、その全 image に対応する FBO を用意する。 */
273 /* 実際に描画する image は毎フレーム xrAcquireSwapchainImage() が決定する。 */
274 auto& swapchain{ impl->swapchains[i] };
275 swapchain.width = configurations[i].recommendedImageRectWidth;
276 swapchain.height = configurations[i].recommendedImageRectHeight;
277 XrSwapchainCreateInfo createInfo{ XR_TYPE_SWAPCHAIN_CREATE_INFO };
278 createInfo.usageFlags = XR_SWAPCHAIN_USAGE_COLOR_ATTACHMENT_BIT | XR_SWAPCHAIN_USAGE_TRANSFER_DST_BIT;
279 createInfo.format = format;
280 createInfo.sampleCount = 1;
281 createInfo.width = swapchain.width;
282 createInfo.height = swapchain.height;
283 createInfo.faceCount = 1;
284 createInfo.arraySize = 1;
285 createInfo.mipCount = 1;
286 if (!impl->check(xrCreateSwapchain(impl->session, &createInfo, &swapchain.handle)))
287 {
288 shutdown();
289 return false;
290 }
291 uint32_t imageCount{};
292 xrEnumerateSwapchainImages(swapchain.handle, 0, &imageCount, nullptr);
293 swapchain.images.assign(imageCount, { XR_TYPE_SWAPCHAIN_IMAGE_OPENGL_KHR });
294 xrEnumerateSwapchainImages(swapchain.handle, imageCount, &imageCount,
295 reinterpret_cast<XrSwapchainImageBaseHeader*>(swapchain.images.data()));
296 swapchain.fbos.resize(imageCount);
297 glGenFramebuffers(static_cast<GLsizei>(imageCount), swapchain.fbos.data());
298 for (uint32_t image{}; image < imageCount; ++image)
299 {
300 /* OpenXR が所有する OpenGL texture をカラー attachment として借用する。 */
301 glBindFramebuffer(GL_FRAMEBUFFER, swapchain.fbos[image]);
302 glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D,
303 swapchain.images[image].image, 0);
304 }
305 impl->publicViews[i].width = static_cast<int>(swapchain.width);
306 impl->publicViews[i].height = static_cast<int>(swapchain.height);
307 }
308 glBindFramebuffer(GL_FRAMEBUFFER, 0);
309 return true;
310#endif
311}

323void GgOpenXR::pollEvents()
324{
325#if defined( GG_ENABLE_OPENXR)
326 if (impl->instance == XR_NULL_HANDLE) return;
327 /* OpenXR の session はイベント駆動で開始・停止する。READY になる前に */
328 /* xrBeginSession() を呼ばず、STOPPING を受けたときだけ xrEndSession() する。 */
329 XrEventDataBuffer event{ XR_TYPE_EVENT_DATA_BUFFER };
330 while (xrPollEvent(impl->instance, &event) == XR_SUCCESS)
331 {
332 if (event.type == XR_TYPE_EVENT_DATA_INSTANCE_LOSS_PENDING) impl->closeRequested = true;
333 else if (event.type == XR_TYPE_EVENT_DATA_SESSION_STATE_CHANGED)
334 {
335 const auto& changed{ *reinterpret_cast<XrEventDataSessionStateChanged*>(&event) };
336 impl->sessionState = changed.state;
337 if (changed.state == XR_SESSION_STATE_READY && !impl->sessionRunning)
338 {
339 XrSessionBeginInfo beginInfo{ XR_TYPE_SESSION_BEGIN_INFO };
340 beginInfo.primaryViewConfigurationType = XR_VIEW_CONFIGURATION_TYPE_PRIMARY_STEREO;
341 impl->sessionRunning = XR_SUCCEEDED(xrBeginSession(impl->session, &beginInfo));
342 }
343 else if (changed.state == XR_SESSION_STATE_STOPPING && impl->sessionRunning)
344 {
345 xrEndSession(impl->session);
346 impl->sessionRunning = false;
347 }
348 else if (changed.state == XR_SESSION_STATE_EXITING || changed.state == XR_SESSION_STATE_LOSS_PENDING)
349 impl->closeRequested = true;
350 }
351 event = { XR_TYPE_EVENT_DATA_BUFFER };
352 }
353#endif
354}

356bool GgOpenXR::beginFrame()
357{
358#if !defined( GG_ENABLE_OPENXR)
359 return false;
360#else
361 /* pollEvents() をここでも呼び、呼び出し側が明示的なイベント取得を忘れても */
362 /* session state が更新されるようにする。 */
363 pollEvents();
364 if (!impl->sessionRunning || impl->frameActive) return false;
365 /* xrWaitFrame() がランタイムとのフレーム周期同期と予測表示時刻の取得を行う。 */
366 /* OpenXR が要求する wait -> begin -> end の順序をこのクラス内で維持する。 */
367 XrFrameWaitInfo waitInfo{ XR_TYPE_FRAME_WAIT_INFO };
368 if (XR_FAILED(xrWaitFrame(impl->session, &waitInfo, &impl->frameState))) return false;
369 XrFrameBeginInfo beginInfo{ XR_TYPE_FRAME_BEGIN_INFO };
370 if (XR_FAILED(xrBeginFrame(impl->session, &beginInfo))) return false;
371 impl->frameActive = true;
372 impl->renderRequested = impl->frameState.shouldRender == XR_TRUE;
373 /* 非表示中でも begin 済みフレームは endFrame() で空レイヤーとして閉じる必要がある。 */
374 if (!impl->renderRequested) return true;
375
376 /* VIEW 空間をアプリケーション基準空間へ locate し、左右眼の中間を推測せず */
377 /* 予測表示時刻における HMD 中央の位置・方向を取得する。 */
378 XrSpaceLocation headLocation{ XR_TYPE_SPACE_LOCATION };
379 if (XR_SUCCEEDED(xrLocateSpace(impl->viewSpace, impl->space,
380 impl->frameState.predictedDisplayTime, &headLocation)) &&
381 (headLocation.locationFlags & XR_SPACE_LOCATION_POSITION_VALID_BIT) &&
382 (headLocation.locationFlags & XR_SPACE_LOCATION_ORIENTATION_VALID_BIT))
383 {
384 const auto& pose{ headLocation.pose };
385 impl->headPose.position = { pose.position.x, pose.position.y, pose.position.z };
386 impl->headPose.orientation = { pose.orientation.x, pose.orientation.y,
387 pose.orientation.z, pose.orientation.w };
388 impl->poseValid = true;
389 }
390 else
391 {
392 impl->poseValid = false;
393 }
394
395 /* 同じ予測表示時刻を使って、描画と projection layer 提出に必要な各眼の */
396 /* pose と非対称 FOV を取得する。 */
397 XrViewLocateInfo locateInfo{ XR_TYPE_VIEW_LOCATE_INFO };
398 locateInfo.viewConfigurationType = XR_VIEW_CONFIGURATION_TYPE_PRIMARY_STEREO;
399 locateInfo.displayTime = impl->frameState.predictedDisplayTime;
400 locateInfo.space = impl->space;
401 XrViewState viewState{ XR_TYPE_VIEW_STATE };
402 uint32_t count{ static_cast<uint32_t>(impl->views.size()) };
403 if (XR_FAILED(xrLocateViews(impl->session, &locateInfo, &viewState, count, &count, impl->views.data())))
404 impl->renderRequested = false;
405 for (uint32_t i{}; i < count; ++i)
406 {
407 const auto& source{ impl->views[i] };
408 auto& target{ impl->publicViews[i] };
409 target.position = { source.pose.position.x, source.pose.position.y, source.pose.position.z };
410 target.orientation = { source.pose.orientation.x, source.pose.orientation.y,
411 source.pose.orientation.z, source.pose.orientation.w };
412 target.fov = { source.fov.angleLeft, source.fov.angleRight,
413 source.fov.angleDown, source.fov.angleUp };
414 }
415 return true;
416#endif
417}

419bool GgOpenXR::beginView(std::size_t view)
420{
421#if !defined( GG_ENABLE_OPENXR)
422 return false;
423#else
424 if (!impl->frameActive || !impl->renderRequested || view >= impl->swapchains.size()) return false;
425 auto& swapchain{ impl->swapchains[view] };
426 /* ランタイムから一枚の image を借り、GPU が書き込み可能になるまで待機する。 */
427 /* acquire に成功した時点から endView() まで、本クラスが返却責任を持つ。 */
428 XrSwapchainImageAcquireInfo acquireInfo{ XR_TYPE_SWAPCHAIN_IMAGE_ACQUIRE_INFO };
429 if (XR_FAILED(xrAcquireSwapchainImage(swapchain.handle, &acquireInfo, &swapchain.imageIndex))) return false;
430 swapchain.acquired = true;
431 XrSwapchainImageWaitInfo waitInfo{ XR_TYPE_SWAPCHAIN_IMAGE_WAIT_INFO };
432 waitInfo.timeout = XR_INFINITE_DURATION;
433 if (XR_FAILED(xrWaitSwapchainImage(swapchain.handle, &waitInfo)))
434 {
435 endView(view);
436 return false;
437 }
438 /* 呼び出し側は以降、通常の OpenGL 描画を行うだけで swapchain へ描画できる。 */
439 glBindFramebuffer(GL_DRAW_FRAMEBUFFER, swapchain.fbos[swapchain.imageIndex]);
440 glViewport(0, 0, swapchain.width, swapchain.height);
441 glClear(GL_COLOR_BUFFER_BIT);
442 return true;
443#endif
444}

446void GgOpenXR::endView(std::size_t view)
447{
448#if defined( GG_ENABLE_OPENXR)
449 if (view >= impl->swapchains.size()) return;
450 auto& swapchain{ impl->swapchains[view] };
451 if (!swapchain.acquired) return;
452 /* image をランタイムへ返す前に OpenGL コマンドを GPU へ送る。 */
453 /* release 後はアプリケーションがこの image を使用してはならない。 */
454 glFlush();
455 XrSwapchainImageReleaseInfo releaseInfo{ XR_TYPE_SWAPCHAIN_IMAGE_RELEASE_INFO };
456 xrReleaseSwapchainImage(swapchain.handle, &releaseInfo);
457 swapchain.acquired = false;
458 glBindFramebuffer(GL_DRAW_FRAMEBUFFER, 0);
459#else
460 (void)view;
461#endif
462}

464bool GgOpenXR::endFrame()
465{
466#if !defined( GG_ENABLE_OPENXR)
467 return false;
468#else
469 if (!impl->frameActive) return false;
470 std::vector<XrCompositionLayerProjectionView> projectionViews;
471 XrCompositionLayerProjection layer{ XR_TYPE_COMPOSITION_LAYER_PROJECTION };
472 std::vector<const XrCompositionLayerBaseHeader*> layers;
473 if (impl->renderRequested)
474 {
475 /* beginFrame() で locate した pose/FOV と各 view の swapchain を一つの */
476 /* projection layer にまとめ、同じ predictedDisplayTime で提出する。 */
477 projectionViews.assign(impl->views.size(), { XR_TYPE_COMPOSITION_LAYER_PROJECTION_VIEW });
478 for (std::size_t i{}; i < projectionViews.size(); ++i)
479 {
480 projectionViews[i].pose = impl->views[i].pose;
481 projectionViews[i].fov = impl->views[i].fov;
482 projectionViews[i].subImage.swapchain = impl->swapchains[i].handle;
483 projectionViews[i].subImage.imageRect.extent =
484 { static_cast<int32_t>(impl->swapchains[i].width), static_cast<int32_t>(impl->swapchains[i].height) };
485 }
486 layer.space = impl->space;
487 layer.viewCount = static_cast<uint32_t>(projectionViews.size());
488 layer.views = projectionViews.data();
489 layers.push_back(reinterpret_cast<const XrCompositionLayerBaseHeader*>(&layer));
490 }
491 /* shouldRender == false の場合は layerCount == 0 の空フレームとして提出する。 */
492 XrFrameEndInfo endInfo{ XR_TYPE_FRAME_END_INFO };
493 endInfo.displayTime = impl->frameState.predictedDisplayTime;
494 endInfo.environmentBlendMode = XR_ENVIRONMENT_BLEND_MODE_OPAQUE;
495 endInfo.layerCount = static_cast<uint32_t>(layers.size());
496 endInfo.layers = layers.data();
497 impl->frameActive = false;
498 return XR_SUCCEEDED(xrEndFrame(impl->session, &endInfo));
499#endif
500}

528gg::GgMatrix GgOpenXR::getHeadPoseMatrix() const
529{
530 /* Pose 自体は OpenXR や gg に依存しない公開形式で保持し、必要な場合だけ */
531 /* アプリケーションで扱いやすい GgMatrix の T * R へ変換する。 */
532 if (!impl->poseValid) return gg::ggIdentity();
533 const auto& position{ impl->headPose.position };
534 const auto& orientation{ impl->headPose.orientation };
535 return gg::ggTranslate(position[0], position[1], position[2]) *
536 gg::GgQuaternion{ orientation[0], orientation[1], orientation[2], orientation[3] }.getMatrix();
